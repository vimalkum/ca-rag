class ReferenceLoader:
    """Handles loading and parsing reference documents"""
    
    @staticmethod
    def load_reference_text(pdf_path: str, titles: List[str]) -> Dict[str, str]:
        """
        Load reference text from PDF and split by titles
        
        Args:
            pdf_path: Path to reference PDF
            titles: List of section titles to extract
            
        Returns:
            Dictionary mapping titles to their content
        """
        logger.info(f"Loading reference document from {pdf_path}")
        
        if not os.path.exists(pdf_path):
            logger.warning(f"Reference PDF not found: {pdf_path}")
            return {title.rstrip(): '' for title in titles}
        
        try:
            loader = PyPDFLoader(pdf_path)
            pages = loader.load_and_split()
            full_text = " ".join([doc.page_content for doc in pages])
            
            ref_text = {}
            remaining_text = full_text
            
            # Process titles in reverse order to extract sections
            for title in reversed(titles):
                clean_title = title.rstrip()
                parts = remaining_text.split(clean_title, 1)
                
                if len(parts) > 1:
                    ref_text[clean_title] = parts[1].strip()
                    remaining_text = parts[0]
                else:
                    ref_text[clean_title] = ''
                    logger.warning(f"Title not found in reference: {clean_title}")
            
            logger.info(f"✓ Loaded {len(ref_text)} reference sections")
            return ref_text
            
        except Exception as e:
            logger.error(f"Failed to load reference PDF: {e}")
            return {title.rstrip(): '' for title in titles}

    def _extract_session_data(self, title: str) -> Optional[Tuple[str, str, List, str]]:
        """
        Safely extract data from Streamlit session state
        
        Args:
            title: Section title
            
        Returns:
            Tuple of (prompt, answer, contexts, ground_truth) or None if missing
        """
        try:
            clean_title = title.rstrip()
            print(st.session_state.generated_output)
            # Check if title exists in session state
            if clean_title not in st.session_state.generated_output:
                logger.warning(f"Title not in generated_output: {clean_title}")
                return None
            
            # Extract data with safety checks
            prompt = st.session_state.prmpt.get(clean_title)
            answer = st.session_state.generated_output.get(clean_title)
            docs = st.session_state.rdocs.get(clean_title, [])
            ground_truth = self.ref_text.get(clean_title, '')
            
            # Validate required fields
            if not prompt or not answer:
                logger.warning(f"Missing prompt or answer for: {clean_title}")
                return None
            
            # Extract contexts from documents
            contexts = [doc.page_content for doc in docs if hasattr(doc, 'page_content')]
            
            if not contexts:
                logger.warning(f"No valid contexts for: {clean_title}")
                return None
            
            return prompt, answer, contexts, ground_truth
            
        except (KeyError, AttributeError) as e:
            logger.warning(f"Error extracting data for '{title}': {e}")
            return None
        except Exception as e:
            logger.error(f"Unexpected error for '{title}': {e}")
            return None
    
    def create_test_dataset(self) -> Dataset:
        """
        Convert test cases to RAGAS-compatible dataset from session state
        
        Returns:
            Dataset object for RAGAS evaluation
        """
        logger.info("Creating test dataset...")
        
        data = {
            'question': [],
            'answer': [],
            'contexts': [],
            'ground_truth': []
        }
        
        processed_count = 0
        skipped_count = 0
        
        for title in self.titles:
            result = self._extract_session_data(title)
            
            if result is None:
                skipped_count += 1
                continue
            
            prompt, answer, contexts, ground_truth = result
            
            data['question'].append(prompt)
            data['answer'].append(answer)
            data['contexts'].append(contexts)
            data['ground_truth'].append(ground_truth)
            
            processed_count += 1
        
        logger.info(f"✓ Dataset created: {processed_count} samples processed, {skipped_count} skipped")
        
        if processed_count == 0:
            raise ValueError("No valid test cases found in session state")
        
        return Dataset.from_dict(data)
    
    def evaluate_pipeline_simple(self) -> Dict[str, Any]:
        """
        Simplified evaluation without batching (use for small datasets < 10 samples)
        
        Returns:
            Dictionary containing all evaluation metrics and detailed results
        """
        logger.info("Starting simple pipeline evaluation (no batching)...")
        
        try:
            dataset = self.create_test_dataset()
            '''
            results = evaluate(
                dataset,
                llm=self.llm,
                metrics=self.metrics,
                embeddings=self.evaluator_embeddings,
                run_config=RunConfig(max_workers=1),
                raise_exceptions=False  # Don't fail on individual errors
            )
            '''
            results = evaluate(
                dataset,
                metrics=self.metrics,
                llm=self.llm
            )
            
            results_df = results.to_pandas()
            
            metric_columns = [
                'faithfulness', 
                'answer_relevancy', 
                'context_precision', 
                'context_recall', 
                'answer_correctness'
            ]
            
            aggregate_metrics = {
                metric: float(results_df[metric].mean()) 
                for metric in metric_columns 
                if metric in results_df.columns
            }
            
            if aggregate_metrics:
                aggregate_metrics['overall_score'] = sum(aggregate_metrics.values()) / len(aggregate_metrics)
            else:
                aggregate_metrics['overall_score'] = 0.0
            
            logger.info(f"✓ Evaluation complete. Overall score: {aggregate_metrics['overall_score']:.4f}")
            
            return {
                'aggregate_metrics': aggregate_metrics,
                'detailed_results': results_df,
                'timestamp': datetime.now().isoformat()
            }
            
        except Exception as e:
            logger.error(f"Evaluation failed: {e}")
            raise
    
    def evaluate_pipeline(
        self, 
        batch_size: Optional[int] = None,
        max_retries: Optional[int] = None,
        timeout_per_sample: Optional[int] = None
    ) -> Dict[str, Any]:
        """
        Run complete evaluation on test cases with timeout handling and batching
        
        Args:
            batch_size: Number of samples to process at once (default from config)
            max_retries: Maximum retry attempts for failed batches (default from config)
            timeout_per_sample: Timeout in seconds per sample (default from config)
            
        Returns:
            Dictionary containing all evaluation metrics and detailed results
        """
        # Use config defaults if not provided
        batch_size = batch_size or self.config.eval_batch_size
        max_retries = max_retries or self.config.eval_max_retries
        timeout_per_sample = timeout_per_sample or self.config.eval_timeout_per_sample
        
        logger.info("Starting pipeline evaluation...")
        logger.info(f"Config: batch_size={batch_size}, max_retries={max_retries}, timeout={timeout_per_sample}s")
        
        try:
            # Create dataset
            dataset = self.create_test_dataset()
            total_samples = len(dataset)
            logger.info(f"Total samples to evaluate: {total_samples}")
            
            # Process in batches to avoid timeout
            all_results = []
            
            for i in range(0, total_samples, batch_size):
                batch_end = min(i + batch_size, total_samples)
                batch_dataset = dataset.select(range(i, batch_end))
                batch_num = (i // batch_size) + 1
                total_batches = (total_samples + batch_size - 1) // batch_size
                
                logger.info(f"Processing batch {batch_num}/{total_batches} (samples {i+1}-{batch_end})")
                
                # Retry logic for each batch
                for attempt in range(max_retries):
                    try:
                        # Run RAGAS evaluation on batch
                        # Note: Using concurrent.futures for better timeout control
                        from concurrent.futures import ThreadPoolExecutor, TimeoutError as FuturesTimeoutError
                        
                        def run_evaluation():
                            '''
                            return evaluate(
                                batch_dataset,
                                llm=self.llm,
                                metrics=self.metrics,
                                embeddings=self.evaluator_embeddings,
                                raise_exceptions=False, run_config=RunConfig(max_workers=1)
                            )
                            '''
                            return evaluate(
                                batch_dataset,
                                metrics=self.metrics,
                                llm = self.llm
                            )
                        
                        # Run with timeout using ThreadPoolExecutor
                        with ThreadPoolExecutor(max_workers=1) as executor:
                            future = executor.submit(run_evaluation)
                            try:
                                batch_results = future.result(timeout=timeout_per_sample * len(batch_dataset))
                                all_results.append(batch_results.to_pandas())
                                logger.info(f"✓ Batch {batch_num} completed successfully")
                                break
                            except FuturesTimeoutError:
                                logger.warning(f"Batch {batch_num} timeout after {timeout_per_sample * len(batch_dataset)}s")
                                raise TimeoutError(f"Evaluation timeout for batch {batch_num}")
                        
                    except (TimeoutError, Exception) as e:
                        
                        if attempt < max_retries - 1:
                            logger.warning(f"Batch {batch_num} failed (attempt {attempt+1}/{max_retries}): {e}")
                            logger.info("Retrying with smaller batch size...")
                            
                            # Try with even smaller batch
                            if len(batch_dataset) > 1:
                                smaller_batch_size = max(1, len(batch_dataset) // 2)
                                for j in range(0, len(batch_dataset), smaller_batch_size):
                                    small_batch_end = min(j + smaller_batch_size, len(batch_dataset))
                                    small_batch = batch_dataset.select(range(j, small_batch_end))
                                    
                                    try:
                                        small_results = evaluate(
                                            small_batch,
                                            metrics=self.metrics,
                                            llm=self.llm
                                        )
                                        all_results.append(small_results.to_pandas())
                                        logger.info(f"✓ Sub-batch {j//smaller_batch_size + 1} completed")
                                    except Exception as sub_e:
                                        logger.error(f"Sub-batch failed: {sub_e}")
                                        continue
                                break
                        else:
                            logger.error(f"Batch {batch_num} failed after {max_retries} attempts: {e}")
                            # Continue with next batch instead of failing completely
                            continue
            
            if not all_results:
                raise ValueError("All evaluation batches failed. Please check your configuration and data.")
            
            # Combine all batch results
            results_df = pd.concat(all_results, ignore_index=True)
            
            # Calculate aggregate metrics
            metric_columns = [
                'faithfulness', 
                'answer_relevancy', 
                'context_precision', 
                'context_recall', 
                'answer_correctness'
            ]
            
            aggregate_metrics = {
                metric: float(results_df[metric].mean()) 
                for metric in metric_columns 
                if metric in results_df.columns
            }
            
            # Calculate overall score
            if aggregate_metrics:
                aggregate_metrics['overall_score'] = sum(aggregate_metrics.values()) / len(aggregate_metrics)
            else:
                aggregate_metrics['overall_score'] = 0.0
            
            logger.info(f"✓ Evaluation complete. Overall score: {aggregate_metrics['overall_score']:.4f}")
            logger.info(f"Successfully evaluated {len(results_df)}/{total_samples} samples")
            
            return {
                'aggregate_metrics': aggregate_metrics,
                'detailed_results': results_df,
                'timestamp': datetime.now().isoformat(),
                'samples_evaluated': len(results_df),
                'total_samples': total_samples
            }
            
        except Exception as e:
            logger.error(f"Evaluation failed: {e}")
            raise
    
    def save_results(
        self, 
        results: Dict[str, Any], 
        filename: Optional[str] = None
    ) -> Tuple[str, str]:
        """
        Save evaluation results to files
        
        Args:
            results: Evaluation results dictionary
            filename: Optional custom filename
            
        Returns:
            Tuple of (metrics_path, csv_path)
        """
        if filename is None:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = f"ragas_evaluation_{timestamp}"
        
        try:
            # Save aggregate metrics as JSON
            metrics_path = os.path.join(self.config.output_dir, f"{filename}_metrics.json")
            with open(metrics_path, 'w') as f:
                json.dump(results['aggregate_metrics'], f, indent=2)
            
            # Save detailed results as CSV
            csv_path = os.path.join(self.config.output_dir, f"{filename}_detailed.csv")
            results['detailed_results'].to_csv(csv_path, index=False)
            
            logger.info(f"✓ Results saved to:\n  - {metrics_path}\n  - {csv_path}")
            
            return metrics_path, csv_path
            
        except Exception as e:
            logger.error(f"Failed to save results: {e}")
            raise

    def compare_evaluations(
        self, 
        results1: Dict[str, Any], 
        results2: Dict[str, Any]
    ) -> str:
        """
        Compare two evaluation runs to track improvements
        
        Args:
            results1: First evaluation results (baseline)
            results2: Second evaluation results (current)
            
        Returns:
            Comparison report string
        """
        m1 = results1['aggregate_metrics']
        m2 = results2['aggregate_metrics']
        
        report = f"""
{'='*60}
EVALUATION COMPARISON
{'='*60}
Baseline:  {results1['timestamp']}
Current:   {results2['timestamp']}

METRIC CHANGES:
{'-'*60}
"""
        for metric in m1.keys():
            if metric == 'overall_score':
                continue
            
            if metric not in m2:
                continue
                
            change = m2[metric] - m1[metric]
            pct_change = (change / m1[metric] * 100) if m1[metric] > 0 else 0
            symbol = "↑" if change > 0 else "↓" if change < 0 else "→"
            report += f"{metric:25s} {symbol} {change:+.4f} ({pct_change:+.2f}%)\n"
        
        overall_change = m2['overall_score'] - m1['overall_score']
        report += f"{'-'*60}\n"
        report += f"Overall Score Change:     {overall_change:+.4f}\n"
        report += f"{'='*60}\n"
        
        return report
